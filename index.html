def test_get_single_user_returns_200_and_expected_fields():
    response = requests.get(f"{BASE_URL}/users/2", headers=HEADERS)
    assert response.status_code == 200
    body = response.json()["data"]
    for field in ("id", "email", "first_name"):
        assert field in body

# Negative: nonexistent id should 404, not a misleading 200
